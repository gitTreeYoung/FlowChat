window.FlowChatAPI 已暴露在页面全局，
可通过 DevTools Console 或 MCP evaluate_script 调用。
发送并等待所有回复
const r = await window.FlowChatAPI.ask(
  "你的问题"
);
// r.responses  — { platform: { text, ts } }
// r.duration   — 总耗时(ms)
仅发送（异步，不等待）
await window.FlowChatAPI.sendMessage(
  "你的问题"
);
指定平台发送
const r = await window.FlowChatAPI.ask(
  "你的问题",
  { platforms: ["claude", "chatgpt"],
    timeout: 60000 }
);
查看当前回复
window.FlowChatAPI.getResponses();
window.FlowChatAPI.getStatus();
等待生成完成
const r = await window.FlowChatAPI
  .waitForCompletion({ timeout: 60000 });
获取对话历史
window.FlowChatAPI.getConversationLog();
